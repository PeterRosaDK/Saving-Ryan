### script chunk=192 name="Dialog" type=3 handlers=1
-- Behavior
global question
global Answer

on Dialog
  sprite(10).blendLevel = 70
  updateStage()
  sprite(11).member = question
  updateStage()
  repeat while sprite(11).movieTime < member(question).duration
    updateStage()
  end repeat
  if Answer <> EMPTY then
    sprite(11).member = Answer
    updateStage()
    repeat while sprite(11).movieTime < member(Answer).duration
      updateStage()
    end repeat
  end if
  sprite(10).blendLevel = 255
  sprite(11).member = member("BlankVideo")
  updateStage()
end
